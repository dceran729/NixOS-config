#!/usr/bin/env bash

ACTION=$1
MAX=$(brightnessctl -d intel_backlight max)
CURRENT_RAW=$(brightnessctl -d intel_backlight get)
CURRENT=$(( CURRENT_RAW * 100 / MAX ))

if [[ "$ACTION" == "up" ]]; then
    if (( CURRENT < 5 )); then
        TARGET=$(( CURRENT + 1 ))
    else
        TARGET=$(( (CURRENT / 5) * 5 + 5 ))
    fi
    (( TARGET > 100 )) && TARGET=100

elif [[ "$ACTION" == "down" ]]; then
    if (( CURRENT <= 5 )); then
        TARGET=$(( CURRENT - 1 ))
    else
        TARGET=$(( (CURRENT / 5) * 5 ))
        if (( TARGET == CURRENT )); then
            TARGET=$(( CURRENT - 5 ))
        fi
    fi
    (( TARGET < 1 )) && TARGET=1
else
    exit 1
fi

# Ustawiamy fizyczną jasność

swayosd-client --brightness "$TARGET"

#!/usr/bin/env bash
# Przełącznik eduroam (PUT, EAP-TLS przez wpa_supplicant).
# Jeśli eduroam jest wyłączone, włącza je (iwd zostaje zatrzymany).
# Jeśli jest włączone, wyłącza je i przywraca iwd.

D=/var/lib/eduroam
IFACE=eduroam0
PIDFILE=/run/eduroam-wpa.pid

# sudo nie widzi programów z profilu użytkownika, więc przekazujemy ich ścieżki
if [ "$EUID" -ne 0 ]; then
  exec sudo env IW="$(command -v iw)" WPA="$(command -v wpa_supplicant)" \
    DHCP="$(command -v dhcpcd)" "$0" "$@"
fi

for v in IW WPA DHCP; do
  if [ ! -x "${!v}" ]; then
    echo "Brak programu ($v). Czy zrobiłeś rebuild po dodaniu skryptu?"
    exit 1
  fi
done

active() { [ -f "$PIDFILE" ] && kill -0 "$(cat "$PIDFILE")" 2>/dev/null; }

off() {
  if [ -f "$PIDFILE" ]; then
    kill "$(cat "$PIDFILE")" 2>/dev/null
    rm -f "$PIDFILE"
  fi
  "$DHCP" -k "$IFACE" 2>/dev/null
  ip link delete "$IFACE" 2>/dev/null
  rm -f /run/eduroam-wpa.conf
  echo "Przywracam iwd..."
  systemctl start iwd
  for _ in $(seq 1 30); do
    ip link show wlan0 >/dev/null 2>&1 && break
    sleep 1
  done
  echo "Gotowe: eduroam wyłączone. NetworkManager połączy się sam, impala działa."
}

on() {
  if [ ! -f "$D/eduroam-client.pem" ] || [ ! -f "$D/eduroam-key.pem" ]; then
    echo "Brak certyfikatów w $D"
    exit 1
  fi

  echo "Zatrzymuję iwd..."
  systemctl stop iwd
  sleep 2

  PHY=$("$IW" phy | awk '/^Wiphy/{print $2; exit}')
  ip link delete "$IFACE" 2>/dev/null
  if ! "$IW" phy "$PHY" interface add "$IFACE" type managed; then
    echo "Nie udało się utworzyć interfejsu $IFACE"
    off
    exit 1
  fi
  nmcli device set "$IFACE" managed no 2>/dev/null
  ip link set "$IFACE" up

  cat > /run/eduroam-wpa.conf <<EOF
network={
  ssid="eduroam"
  key_mgmt=WPA-EAP
  eap=TLS
  identity="damian.ceran@student.put.poznan.pl"
  client_cert="$D/eduroam-client.pem"
  private_key="$D/eduroam-key.pem"
}
EOF
  chmod 600 /run/eduroam-wpa.conf

  echo "Łączenie z eduroam..."
  "$WPA" -B -i "$IFACE" -c /run/eduroam-wpa.conf -P "$PIDFILE" -f /run/eduroam-wpa.log

  for _ in $(seq 1 30); do
    [ "$(cat /sys/class/net/$IFACE/carrier 2>/dev/null)" = 1 ] && break
    sleep 1
  done

  if [ "$(cat /sys/class/net/$IFACE/carrier 2>/dev/null)" != 1 ]; then
    echo "Nie udało się uwierzytelnić. Ostatnie linie logu:"
    tail -n 15 /run/eduroam-wpa.log 2>/dev/null
    off
    exit 1
  fi

  "$DHCP" -q "$IFACE"
  sleep 2
  ip -br addr show "$IFACE"
  echo "Gotowe: eduroam włączone (iwd wyłączony). Uruchom skrypt ponownie, żeby wyłączyć."
}

if active; then off; else on; fi

#!/usr/bin/env bash

dir="$HOME/.config/rofi"

# CMDs
uptime="`uptime -p | sed -e 's/up //g'`"
host=`hostname`

# Options
shutdown=''
reboot=''
logout=''

# Rofi CMD
rofi_cmd() {
    rofi -dmenu \
        -p "Uptime: $uptime" \
        -mesg "Uptime: $uptime" \
        -theme ${dir}/powermenu.rasi
}

# Pass variables to rofi dmenu
run_rofi() {
    echo -e "$logout\n$reboot\n$shutdown" | rofi_cmd
}

# Actions
chosen="$(run_rofi)"
case ${chosen} in
    $shutdown)
        systemctl poweroff
        ;;
    $reboot)
        systemctl reboot
        ;;
    $logout)
        hyprctl dispatch 'hl.dsp.exit()'
        ;;
esac

========================================
PLIK: hosts/server/qbit-cleanup.nix
========================================
{ config, pkgs, ... }:

{
  systemd.services.qbit-cleanup = {
    description = "Usuwa martwe torrenty przez Sonarr/Radarr (blocklist + ponowne wyszukiwanie)";
    after = [ "qbittorrent.service" "sonarr.service" "radarr.service" ];
    path = [
      pkgs.curl
      pkgs.jq
      pkgs.coreutils
    ];

    serviceConfig = {
      Type = "oneshot";
      # Plik /var/lib/secrets/qbit-cleanup.env (chmod 600, własność roota):
      #   QBIT_USER=...
      #   QBIT_PASS=...
      #   SONARR_API_KEY=...   (Sonarr -> Settings -> General -> API Key)
      #   RADARR_API_KEY=...   (Radarr -> Settings -> General -> API Key)
      EnvironmentFile = "/var/lib/secrets/qbit-cleanup.env";
    };

    script = ''
      QBIT_URL="http://127.0.0.1:8080"
      SONARR_URL="http://127.0.0.1:8989"
      RADARR_URL="http://127.0.0.1:7878"

      # metaDL: brak metadanych dłużej niż 2 min od dodania
      # stalledDL: brak jakiejkolwiek aktywności od 2 min
      META_TIMEOUT=300
      STALL_TIMEOUT=180
      # Minimalna liczba węzłów DHT, poniżej której qBit uznajemy za "jeszcze się rozkręca"
      MIN_DHT_NODES=10

      COOKIE_JAR=$(mktemp)
      trap 'rm -f "$COOKIE_JAR"' EXIT

      curl -s -c "$COOKIE_JAR" \
        --data-urlencode "username=$QBIT_USER" \
        --data-urlencode "password=$QBIT_PASS" \
        "$QBIT_URL/api/v2/auth/login" > /dev/null

      DHT=$(curl -sf -b "$COOKIE_JAR" "$QBIT_URL/api/v2/transfer/info" | jq -r '.dht_nodes // 0')
      if [ "''${DHT:-0}" -lt "$MIN_DHT_NODES" ]; then
        echo "DHT niegotowe ($DHT węzłów), pomijam czyszczenie"
        curl -s -b "$COOKIE_JAR" "$QBIT_URL/api/v2/auth/logout" > /dev/null
        exit 0
      fi

      JSON=$(curl -sf -b "$COOKIE_JAR" "$QBIT_URL/api/v2/torrents/info") || {
        echo "Nie udało się pobrać listy torrentów z qBittorrenta"
        exit 1
      }
      NOW=$(date +%s)

      HASHES=$(echo "$JSON" | jq -r \
        --argjson now "$NOW" \
        --argjson meta "$META_TIMEOUT" \
        --argjson stall "$STALL_TIMEOUT" '
          .[] | select(
            ((.state == "metaDL" or .state == "forcedMetaDL") and ($now - .added_on > $meta))
            or
            (.state == "stalledDL" and ($now - ([.added_on, .last_activity] | max) > $stall))
          ) | .hash')

      [ -z "$HASHES" ] && { curl -s -b "$COOKIE_JAR" "$QBIT_URL/api/v2/auth/logout" > /dev/null; exit 0; }

      SONARR_QUEUE=$(curl -sf -H "X-Api-Key: $SONARR_API_KEY" "$SONARR_URL/api/v3/queue?pageSize=1000" || echo '{"records":[]}')
      RADARR_QUEUE=$(curl -sf -H "X-Api-Key: $RADARR_API_KEY" "$RADARR_URL/api/v3/queue?pageSize=1000" || echo '{"records":[]}')

      # Usuwa pozycję z kolejki *arr: kasuje z klienta, dodaje do blocklisty
      # i uruchamia ponowne wyszukiwanie (skipRedownload=false).
      # Argumenty: URL, klucz API, JSON kolejki, hash torrenta
      remove_via_arr() {
        id=$(echo "$3" | jq -r --arg h "$4" \
          '[.records[] | select((.downloadId // "" | ascii_downcase) == ($h | ascii_downcase)) | .id][0] // empty')
        [ -z "$id" ] && return 1
        curl -sf -X DELETE -H "X-Api-Key: $2" \
          "$1/api/v3/queue/$id?removeFromClient=true&blocklist=true&skipRedownload=false" > /dev/null
      }

      for hash in $HASHES; do
        if remove_via_arr "$SONARR_URL" "$SONARR_API_KEY" "$SONARR_QUEUE" "$hash"; then
          echo "Sonarr: usunięto i zablokowano $hash"
        elif remove_via_arr "$RADARR_URL" "$RADARR_API_KEY" "$RADARR_QUEUE" "$hash"; then
          echo "Radarr: usunięto i zablokowano $hash"
        else
          # Torrent nie należy do żadnego *arr (dodany ręcznie) - zwykłe usunięcie
          echo "Brak w kolejce *arr, usuwam bezpośrednio: $hash"
          curl -s -b "$COOKIE_JAR" --data "hashes=$hash&deleteFiles=true" \
            "$QBIT_URL/api/v2/torrents/delete" > /dev/null
        fi
      done

      curl -s -b "$COOKIE_JAR" "$QBIT_URL/api/v2/auth/logout" > /dev/null
    '';
  };

  systemd.timers.qbit-cleanup = {
    wantedBy = [ "timers.target" ];
    timerConfig = {
      OnBootSec = "5m";
      OnUnitActiveSec = "1m";
    };
  };
}
