```nix
{ config, pkgs, ... }:

{
  systemd.services.qbit-cleanup = {
    description = "Usuwa martwe torrenty z qBittorrenta by wymusic pobranie kolejnych w Radarr";
    path = [ pkgs.curl pkgs.jq pkgs.coreutils ];
    script = ''
      COOKIE_JAR=$(mktemp)
      QBIT_URL="http://100.95.221.38:8080"

      USER="DCSRW"
      PASS="qwe321"

      curl -s -c "$COOKIE_JAR" --data "username=$USER&password=$PASS" "$QBIT_URL/api/v2/auth/login" > /dev/null
      JSON=$(curl -s -b "$COOKIE_JAR" "$QBIT_URL/api/v2/torrents/info")
      NOW=$(date +%s)

      if [ -n "$JSON" ] && [ "$JSON" != "null" ]; then
        HASHES=$(echo "$JSON" | jq -r --argjson now "$NOW" '.[] | select((.state == "metaDL" or .state == "stalledDL") and ($now - .added_on > 900)) | .hash')
        for hash in $HASHES; do
          curl -s -b "$COOKIE_JAR" --data "hashes=$hash&deleteFiles=true" "$QBIT_URL/api/v2/torrents/delete" > /dev/null
        done
      fi

      curl -s -b "$COOKIE_JAR" "$QBIT_URL/api/v2/auth/logout" > /dev/null
      rm -f "$COOKIE_JAR"
    '';
  };

  systemd.timers.qbit-cleanup = {
    wantedBy = [ "timers.target" ];
    timerConfig = {
      OnBootSec = "5m";
      OnUnitActiveSec = "5m";
    };
  };
}
```

### Plik: `assets/Scripts/brightness.sh`
```nix
#!/usr/bin/env bash

ACTION=$1
MAX=$(brightnessctl -d intel_backlight max)
CURRENT_RAW=$(brightnessctl -d intel_backlight get)
CURRENT=$(( CURRENT_RAW * 100 / MAX ))

if [[ "$ACTION" == "up" ]]; then
    if (( CURRENT < 5 )); then
        TARGET=$(( CURRENT + 1 ))
    else
        TARGET=$(( (CURRENT / 5) * 5 + 5 ))
    fi
    (( TARGET > 100 )) && TARGET=100

elif [[ "$ACTION" == "down" ]]; then
    if (( CURRENT <= 5 )); then
        TARGET=$(( CURRENT - 1 ))
    else
        TARGET=$(( (CURRENT / 5) * 5 ))
        if (( TARGET == CURRENT )); then
            TARGET=$(( CURRENT - 5 ))
        fi
    fi
    (( TARGET < 1 )) && TARGET=1
else
    exit 1
fi

# Ustawiamy fizyczną jasność

swayosd-client --brightness "$TARGET"
```

```nix
fi
```

### Plik: `assets/Scripts/powermenu.sh`
```nix
#!/usr/bin/env bash

dir="$HOME/.config/rofi"

# CMDs
uptime="`uptime -p | sed -e 's/up //g'`"
host=`hostname`

# Options
shutdown=''
reboot=''
logout=''

# Rofi CMD
rofi_cmd() {
    rofi -dmenu \
        -p "Uptime: $uptime" \
        -mesg "Uptime: $uptime" \
        -theme ${dir}/powermenu.rasi
}

# Pass variables to rofi dmenu
run_rofi() {
    echo -e "$logout\n$reboot\n$shutdown" | rofi_cmd
}

# Actions
chosen="$(run_rofi)"
case ${chosen} in
    $shutdown)
        systemctl poweroff
        ;;
    $reboot)
        systemctl reboot
        ;;
    $logout)
        hyprctl dispatch exit
        ;;
esac
```

### Plik: `assets/Scripts/launch-walker.sh`

```nix
#!/usr/bin/env bash

if ! pgrep -x elephant > /dev/null; then
  setsid elephant &
fi

if ! pgrep -f "walker --gapplication-service" > /dev/null; then
  setsid walker --gapplication-service &
```
